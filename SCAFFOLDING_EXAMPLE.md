#!/usr/bin/env python3
"""
Config Manager Entry Point
Generated by Control Flow Engine from design_specs/control_flows.yml

Usage:
    python3 run.py                    # Run main_config_flow
    python3 run.py --flow discover    # Run specific flow
    python3 run.py --help             # Show help
"""

import sys
import argparse
import yaml
from pathlib import Path
from typing import Dict, Any


class ConfigManagerRunner:
    """Executes config-manager flows defined in control_flows.yml"""
    
    def __init__(self):
        self.repo_root = Path(__file__).parent
        self.spec_path = self.repo_root / "design_specs" / "control_flows.yml"
        self.spec = None
        
        # Generated: 2025-10-13 by Control Flow Engine v0.1.0
        sys.path.insert(0, str(self.repo_root))
    
    def load_spec(self) -> Dict[str, Any]:
        """Load control_flows.yml specification."""
        if not self.spec_path.exists():
            print(f"ERROR: Control flow specification not found: {self.spec_path}")
            sys.exit(1)
        
        with open(self.spec_path) as f:
            self.spec = yaml.safe_load(f)
        
        return self.spec
    
    def execute_flow(self, flow_name: str) -> bool:
        """Execute a flow by name from control_flows.yml"""
        if not self.spec:
            self.load_spec()
        
        flows = self.spec.get('flows', {})
        if flow_name not in flows:
            print(f"ERROR: Flow '{flow_name}' not found")
            print(f"Available: {', '.join(flows.keys())}")
            return False
        
        flow = flows[flow_name]
        phases = flow.get('phases', [])
        
        # Execute each phase in sequence
        context = {}
        
        for phase in phases:
            phase_id = phase['phase_id']
            phase_name = phase['name']
            
            print(f"\n→ Phase: {phase_name}")
            
            success = self._execute_phase(phase_id, phase_name, context)
            if not success:
                print(f"ERROR: Phase '{phase_name}' failed")
                return False
            
            print(f"✓ Phase '{phase_name}' completed")
        
        return True
    
    def _execute_phase(self, phase_id: str, phase_name: str, context: Dict[str, Any]) -> bool:
        """Execute a single phase."""
        try:
            phase_module = self._import_phase_module(phase_id)
            phase_class = self._get_phase_class(phase_module, phase_id)
            
            phase_instance = phase_class(project_root=self.repo_root)
            result = phase_instance.execute(context)
            
            if result:
                context.update(result)
            
            return True
        except NotImplementedError:
            print(f"NOTICE: Phase '{phase_name}' needs implementation")
            return False
        except Exception as e:
            print(f"ERROR: {e}")
            return False
    
    def _import_phase_module(self, phase_id: str):
        """Import phase module dynamically."""
        phase_map = {
            'discovery': 'phases.phase_1_discovery',
            'tui_mapping': 'phases.phase_2_tui_mapping',
            'collection': 'phases.phase_3_collection',
            'validation': 'phases.phase_4_validation',
            'export': 'phases.phase_5_export',
        }
        
        module_name = phase_map.get(phase_id)
        return __import__(module_name, fromlist=['*'])
    
    def _get_phase_class(self, phase_module, phase_id: str):
        """Get phase class from module."""
        class_map = {
            'discovery': 'DiscoveryPhase',
            'tui_mapping': 'TUIMappingPhase',
            'collection': 'CollectionPhase',
            'validation': 'ValidationPhase',
            'export': 'ExportPhase',
        }
        
        class_name = class_map.get(phase_id)
        return getattr(phase_module, class_name, None)


def main():
    """Main entry point."""
    parser = argparse.ArgumentParser(description='Config Manager')
    parser.add_argument('--flow', default='main_config_flow')
    parser.add_argument('--list', action='store_true')
    
    args = parser.parse_args()
    
    runner = ConfigManagerRunner()
    runner.load_spec()
    
    if args.list:
        print("\nAvailable flows:")
        for flow_name, flow_data in runner.spec.get('flows', {}).items():
            print(f"  - {flow_name}: {flow_data.get('description')}")
        sys.exit(0)
    
    success = runner.execute_flow(args.flow)
    sys.exit(0 if success else 1)


if __name__ == "__main__":
    main()
