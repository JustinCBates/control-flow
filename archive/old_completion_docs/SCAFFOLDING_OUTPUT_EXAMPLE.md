### 4. Root Entry Point

**Entry Point**: `run.py`

```python
#!/usr/bin/env python3
"""
Data Processor Entry Point
Generated by Control Flow Engine from examples/data_processor_flow.yml
"""

import sys
import argparse
import yaml
from pathlib import Path
from typing import Dict, Any


class DataProcessorRunner:
    """Executes data-processor flows defined in control_flows.yml"""
    
    def __init__(self):
        self.repo_root = Path(__file__).parent
        self.spec_path = self.repo_root / "design_specs" / "control_flows.yml"
        self.spec = None
        
        sys.path.insert(0, str(self.repo_root))
    
    def load_spec(self) -> Dict[str, Any]:
        """Load control_flows.yml specification."""
        with open(self.spec_path) as f:
            self.spec = yaml.safe_load(f)
        return self.spec
    
    def execute_flow(self, flow_name: str) -> bool:
        """Execute a flow by name."""
        if not self.spec:
            self.load_spec()
        
        flows = self.spec.get('flows', {})
        flow = flows.get(flow_name)
        
        if not flow:
            print(f"ERROR: Flow '{flow_name}' not found")
            return False
        
        context = {}
        
        for phase_data in flow.get('phases', []):
            phase_id = phase_data['phase_id']
            phase_name = phase_data['name']
            entry_file = phase_data.get('entry_file', f"{phase_id}.py")
            
            print(f"\n→ Phase: {phase_name}")
            
            # Dynamically import phase
            success = self._execute_phase(phase_id, entry_file, context)
            
            if not success:
                print(f"ERROR: Phase '{phase_name}' failed")
                return False
            
            print(f"✓ Phase '{phase_name}' completed")
        
        return True
    
    def _execute_phase(self, phase_id: str, entry_file: str, 
                      context: Dict[str, Any]) -> bool:
        """Execute a single phase."""
        try:
            # Import phase module
            module_name = entry_file.replace('.py', '')
            phase_module = __import__(
                f"phases.phase_1_{phase_id}.{module_name}",
                fromlist=['*']
            )
            
            # Get phase class
            phase_class = getattr(phase_module, f"{phase_id.title()}Phase")
            
            # Execute
            phase_instance = phase_class(project_root=self.repo_root)
            result = phase_instance.execute(context)
            
            if result:
                context.update(result)
            
            return True
            
        except Exception as e:
            print(f"ERROR: {e}")
            import traceback
            traceback.print_exc()
            return False


def main():
    """Main entry point."""
    parser = argparse.ArgumentParser(description='Data Processor')
    parser.add_argument('--flow', default='main_process_flow')
    parser.add_argument('--list', action='store_true')
    
    args = parser.parse_args()
    
    runner = DataProcessorRunner()
    runner.load_spec()
    
    if args.list:
        print("\nAvailable flows:")
        for flow_name, flow_data in runner.spec.get('flows', {}).items():
            print(f"  - {flow_name}: {flow_data.get('description')}")
        sys.exit(0)
    
    success = runner.execute_flow(args.flow)
    sys.exit(0 if success else 1)


if __name__ == "__main__":
    main()
```
